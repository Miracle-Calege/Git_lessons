@echo off
goto test

//for... do-> commando usado para loop

for %%n in (x,y,z) do echo %%n;

quer dizer: para cada n em (x,y,z) faça comando
n assume os valores entre parenteses.

ex:
@echo off
for %%n in (notepad.exe,notepad.exe) do start %%n
pause>nul

ex2:

/l-> permite o for trabalhar com valores numericos
(0,1,10) 0-valor inicial, 1-o valor de incremento, 10-o valor final que n pode assumir 
@echo off
for /L %%n in (0,1,10) do echo %%n
pause>nul
